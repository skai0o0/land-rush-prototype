Hệ Thống Đặc Tả Kiến Trúc 3D Và Tham Số Procedural Cho Tổng Hành Dinh 7 Trường ĐHQG-HCM Trong Game Chiếm Đất
Quy chuẩn tạo hình tổng hành dinh (Base HQ) cho 7 trường thành viên Đại học Quốc gia Thành phố Hồ Chí Minh (ĐHQG-HCM) trong một tựa game chiến thuật chiếm đất (Territory Control RTS) đòi hỏi việc chuyển đổi các thực thể kiến trúc đời thực thành những công trình viễn tưởng chiến lược (Stylized Sci-Fi RTS Architecture). Mỗi đại bản doanh đóng vai trò là trung tâm tái sinh quân số, trạm điều phối công nghệ và cột mốc trực quan xác lập chủ quyền lãnh thổ. Để đảm bảo tính phân cấp thị giác khi người chơi quan sát từ camera đẳng cự góc nghiêng   , cấu trúc hình học của mọi đại bản doanh được thiết kế theo tỷ lệ khối bao ngoài (Bounding Box) tiêu chuẩn    trên mặt đất và chiều cao vươn từ    đến   . Hệ thống phân rã thành bốn tầng hình học lắp ghép: Chân đế công trình (Sub-structure Plinth), Khối thân điều phối (Core Habitation), Điểm nhấn biểu trưng trên đỉnh (Apex Sigil), và Các phân khu phụ trợ ngoại vi (Functional Modules).
Mô hình kỹ thuật điều khiển qua giao thức Model Context Protocol (MCP) cho phép AI Agent giao tiếp trực tiếp với không gian bộ nhớ của Blender thông qua API Python (bpy). Bằng cách giải trừ các chi tiết vụn vặt và tập trung vào các khối hình học nguyên thủy (Primitives: Box, Cylinder, Torus, UV/Ico Sphere) kết hợp với các phép toán Boole (Boolean Operations) và vát cạnh Bevel góc   , AI Agent có thể kích hoạt các đoạn mã Python để tái tạo tức thì các mô hình có độ tối ưu cao (   đa giác), hỗ trợ tự động gán vật liệu PBR theo màu sắc nhận diện chính thức và định hình ranh giới lãnh thổ một cách cơ động.
Bản sắc 7 phe phái học thuật được phân bổ thành 7 phong cách công trình đặc thù: Trường Đại học Bách khoa mang hình dáng pháo đài cơ khí Brutalist với kết cấu tổ ong lục giác hạng nặng; Trường Đại học Khoa học Tự nhiên định hình như một trạm quan sát lượng tử với hệ vành đai nguyên tử chuyển động; Trường Đại học Khoa học Xã hội và Nhân văn tái hiện một đền đài văn hiến với tam cấp Ziggurat và lầu Khuê Văn Các; Trường Đại học Công nghệ Thông tin sở hữu khối tháp máy chủ Monolith nguyên khối dát mạch vi cơ điện tử; Trường Đại học Kinh tế - Luật uy nghiêm với khối bán nguyệt nghị viện và tháp đôi cán cân kết nối qua cầu trái tim tri thức; Trường Đại học Quốc tế rực rỡ với tổ hợp khối hộp mô-đun đỏ son liên kết bằng ống skywalk; và Trường Đại học Khoa học Sức khỏe thanh khiết với mặt bằng chữ thập cứu thương bao bọc bởi khung xương xoắn kép DNA.

import bpy
import math

def srgb_to_linear(c_srgb):
   """Giải mã gamma sRGB sang Linear sRGB cho hệ thống Shading của Blender."""
   if c_srgb <= 0.04045:
       return c_srgb / 12.92
   else:
       return math.pow((c_srgb + 0.055) / 1.055, 2.4)

def hex_to_linear_rgb(hex_str, alpha=1.0):
   """Chuyển đổi chuỗi Hex thành vector RGBA tuyến tính."""
   hex_clean = hex_str.lstrip('#')
   r = int(hex_clean[0:2], 16) / 255.0
   g = int(hex_clean[2:4], 16) / 255.0
   b = int(hex_clean[4:6], 16) / 255.0
   return (srgb_to_linear(r), srgb_to_linear(g), srgb_to_linear(b), alpha)

def create_pbr_material(name, base_hex, metallic=0.0, roughness=0.5, emit_hex=None, emit_strength=0.0):
   """Khởi tạo vật liệu Principled BSDF với chuẩn kết xuất PBR hiện đại."""
   mat = bpy.data.materials.get(name)
   if mat is None:
       mat = bpy.data.materials.new(name=name)
       mat.use_nodes = True
       
   nodes = mat.node_tree.nodes
   bsdf = nodes.get("Principled BSDF")
   
   if bsdf:
       bsdf.inputs["Base Color"].default_value = hex_to_linear_rgb(base_hex)
       bsdf.inputs["Metallic"].default_value = metallic
       bsdf.inputs["Roughness"].default_value = roughness
       if emit_hex and emit_strength > 0.0:
           bsdf.inputs["Emission Color"].default_value = hex_to_linear_rgb(emit_hex)
           bsdf.inputs["Emission Strength"].default_value = emit_strength
           
   return mat

def build_hcmut_hq(collection):
   """Hàm tham chiếu: Khởi tạo mô hình Tổng hành dinh Đại học Bách khoa (HCMUT)."""
   # Khởi tạo bảng vật liệu
   mat_concrete = create_pbr_material("MAT_BK_Concrete", "#4A5568", metallic=0.1, roughness=0.85)
   mat_armor = create_pbr_material("MAT_BK_ArmorNavy", "#030391", metallic=0.65, roughness=0.35)
   mat_glow = create_pbr_material("MAT_BK_GlowCyan", "#000000", emit_hex="#1488D8", emit_strength=10.0)

   # 1. Chân đế pháo đài lục giác
   bpy.ops.mesh.primitive_cylinder_add(vertices=6, radius=22.0, depth=4.0, location=(0.0, 0.0, 2.0))
   base_obj = bpy.context.active_object
   base_obj.name = "HCMUT_Plinth"
   base_obj.data.materials.append(mat_concrete)
   collection.objects.link(base_obj)
   bpy.context.scene.collection.objects.unlink(base_obj)

   # 2. Tháp chính Hex-Core
   bpy.ops.mesh.primitive_cylinder_add(vertices=6, radius=14.0, depth=24.0, location=(0.0, 0.0, 16.0))
   core_obj = bpy.context.active_object
   core_obj.name = "HCMUT_HexCore"
   core_obj.data.materials.append(mat_armor)
   collection.objects.link(core_obj)
   bpy.context.scene.collection.objects.unlink(core_obj)

   # 3. Buồng phản ứng phát quang
   bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=6.0, depth=8.0, location=(0.0, 0.0, 32.0))
   reactor_obj = bpy.context.active_object
   reactor_obj.name = "HCMUT_Reactor"
   reactor_obj.data.materials.append(mat_glow)
   collection.objects.link(reactor_obj)
   bpy.context.scene.collection.objects.unlink(reactor_obj)

   # Thiết lập phân cấp quan hệ
   core_obj.parent = base_obj
   reactor_obj.parent = base_obj

   return base_obj

def execute_pipeline():
   """Điểm kích hoạt chính từ giao thức MCP."""
   col_name = "VNU_Base_HQ_Collection"
   hq_collection = bpy.data.collections.get(col_name)
   if not hq_collection:
       hq_collection = bpy.data.collections.new(col_name)
       bpy.context.scene.collection.children.link(hq_collection)
       
   build_hcmut_hq(hq_collection)

execute_pipeline()
